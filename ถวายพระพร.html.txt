<script>
function submitForm() {
    const fname = document.getElementById("fname").value;
    const lname = document.getElementById("lname").value;
    const position = document.getElementById("position").value;

    const checkboxes = document.querySelectorAll('input[type="checkbox"]:checked');
    let wishes = [];

    checkboxes.forEach((cb) => {
        wishes.push(cb.value);
    });

    if (!fname || !lname || wishes.length === 0) {
        alert("กรุณากรอกข้อมูลให้ครบ และเลือกคำถวายพระพรอย่างน้อย 1 รายการ");
        return;
    }

    const resultDiv = document.getElementById("result");

    resultDiv.style.display = "block";
    resultDiv.innerHTML = `
        <h3>ข้อมูลการถวายพระพร</h3>
        <p><strong>ชื่อ:</strong> ${fname} ${lname}</p>
        <p><strong>ตำแหน่ง:</strong> ${position}</p>
        <p><strong>คำถวายพระพร:</strong></p>
        <ul>${wishes.map(w => `<li>${w}</li>`).join('')}</ul>
        <p style="color:green;"><strong>ลงทะเบียนถวายพระพรออนไลน์เรียบร้อยแล้ว</strong></p>
    `;
}
</script>
